
// Presentación clara de fecha civil + Oná + fecha hebrea.
// Cambio exclusivamente visual: no modifica la fecha interna, eventos ni cálculos halájicos.
(function clearOnaDateSheet(){
  function formatCivil(d){
    return new Date(d).toLocaleDateString('es-MX',{weekday:'long',day:'numeric',month:'long'});
  }
  function hebDate(d){
    const h=hebParts(new Date(d));
    return h ? (h.day+' de '+hMonthEs(h.month)+' de '+h.year) : heb(new Date(d));
  }
  function onaLabel(o,alerts){
    return (alerts?'Avisos de ':'')+(o==='yom'?'Onat Yom':'Onat Laila');
  }

  const baseOpenSheet=openSheet;
  openSheet=function(d,o){
    const result=baseOpenSheet.apply(this,arguments);
    const title=document.getElementById('selTitle');
    const sub=document.getElementById('selSub');
    if(title)title.textContent=formatCivil(d);
    if(sub)sub.textContent=onaLabel(o,false)+' · '+hebDate(d);
    return result;
  };

  const baseOpenAlertDetails=openAlertDetails;
  openAlertDetails=function(d,o,alerts){
    const result=baseOpenAlertDetails.apply(this,arguments);
    const title=document.getElementById('selTitle');
    const sub=document.getElementById('selSub');
    if(title)title.textContent=formatCivil(d);
    if(sub)sub.textContent=onaLabel(o,true)+' · '+hebDate(d);
    return result;
  };
})();
