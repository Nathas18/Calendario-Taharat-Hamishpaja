function removeReiya(ev){
 // Al eliminar una Reiyá también se elimina el Hefsek perteneciente a ese ciclo.
 // Shivá Nekiim y Tevilá son automáticos a partir del Hefsek, por lo que desaparecen al mismo tiempo.
 const reiyot=state.events.filter(e=>e.type==='reiya').sort((a,b)=>a.date.localeCompare(b.date));
 const idx=reiyot.findIndex(r=>(ev.id&&r.id===ev.id)||(!ev.id&&r.date===ev.date&&r.ona===ev.ona));
 const nextReiya=idx>=0 && idx<reiyot.length-1 ? reiyot[idx+1] : null;
 const targetDate=ev.date || (selected&&selected.date);
 const targetId=ev.id;

 state.events=state.events.filter(e=>{
   // 1) eliminar la propia Reiyá
   if(e.type==='reiya'){
     if(targetId && e.id===targetId) return false;
     if(e.date===targetDate && e.ona===ev.ona) return false;
   }
   // 2) eliminar Hefsek ligado explícitamente a esta Reiyá
   if(e.type==='hefsek' && targetId && e.reiyaId===targetId) return false;
   // 3) respaldo para registros viejos: Hefsek posterior a esta Reiyá y anterior a la siguiente Reiyá
   if(e.type==='hefsek' && !e.reiyaId && targetDate && e.date>=targetDate && (!nextReiya || e.date<nextReiya.date)) return false;
   return true;
 });
 save();
 document.getElementById('modal').style.display='none';
 selected=null;
 render();
}

 msg.textContent=mode==='signup'?'Creando cuenta…':'Iniciando sesión…';
 try{
   const res=mode==='signup'?await authSignUp(email,password):await authSignIn(email,password);

 }catch(error){msg.textContent=error.message||'No se pudo completar la operación.';}
}
