 function injectIfPolanco(){
  const view=document.getElementById('tevilotView');if(!view||view.hidden)return;
  const loc=view.querySelector('.contactLocation');
  if(!loc||loc.textContent.trim()!=='Polanco')return;
  const grid=view.querySelector('.contactGrid');if(!grid)return;
  if(!grid.querySelector('[data-edgar-allan-poe-tevila]'))grid.insertAdjacentHTML('beforeend',cardHtml);
 }
 function wire(){
  const view=document.getElementById('tevilotView'),tab=document.getElementById('tabTevilot');if(!view||!tab)return false;
  if(!tab.dataset.edgarPolancoWired){tab.dataset.edgarPolancoWired='1';tab.addEventListener('click',()=>setTimeout(()=>{wirePolancoButtons();injectIfPolanco()},80));}
  wirePolancoButtons();injectIfPolanco();return true;
 }
 function wirePolancoButtons(){
  const view=document.getElementById('tevilotView');if(!view)return;
  view.querySelectorAll('[data-tevila-place="Polanco"]').forEach(b=>{
   if(b.dataset.edgarPolancoWired)return;b.dataset.edgarPolancoWired='1';
   b.addEventListener('click',()=>setTimeout(injectIfPolanco,80));
  });
 }
 let tries=0;const timer=setInterval(()=>{tries++;if(wire()||tries>40)clearInterval(timer)},100);

 const observer=new MutationObserver(()=>{wirePolancoButtons();setTimeout(injectIfPolanco,0)});
