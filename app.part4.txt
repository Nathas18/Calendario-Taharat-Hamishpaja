.join(' → ')+'. El salto se repite '+(d1>0?'+':'')+d1+' cada mes. Próximo día proyectado: '+target+'.',nx,x[2].ona,{targetDay:target,delta:d1});}
     }
   }
 }
 if(re.length>=4){
   let x=re.slice(-4),hp=x.map(r=>hebParts(atNoon(r.date)));
   if(hp.every(Boolean)&&sameOna(x)&&consecutiveHebMonths(x)){
     let days=hp.map(z=>z.day),ds=[days[1]-days[0],days[2]-days[1],days[3]-days[2]];
     let progressiveForward=ds[0]===1&&ds[1]===2&&ds[2]===3;
     let progressiveBack=ds[0]===-1&&ds[1]===-2&&ds[2]===-3;
     if(progressiveForward||progressiveBack){let nextDelta=progressiveForward?4:-4,target=days[3]+nextDelta,nx=immediateNextHebMonthTarget(x[3].date,target);if(nx){let dir=progressiveForward?'Kadima':'Lemafreá';add('jodesh-dilug-progressive-'+(progressiveForward?'kadima':'lemafre'),'Posible Veset HaJódesh Bedilug '+dir+' · Yamim','Secuencia progresiva de días hebreos en la misma oná: '+days.join(' → ')+'. Los saltos fueron '+ds.map(v=>(v>0?'+':'')+v).join(', ')+'. Próximo salto proyectado: '+(nextDelta>0?'+':'')+nextDelta+' → día '+target+'.',nx,x[3].ona,{targetDay:target,delta:nextDelta,progressive:true});}}
   }
 }
 // Veset Haguf: mismo síntoma 24 h antes de 3 Reiyot. Se conserva como detector separado.
 let symptomHits=[];for(const r of re){let prev=atNoon(r.date);prev.setDate(prev.getDate()-1);let pi=localISO(prev);for(const e of state.events.filter(e=>e.type==='symptom'&&e.date===pi)){symptomHits.push({reiya:r,s:e});}}
 if(symptomHits.length>=3){let z=symptomHits.slice(-3),names=z.map(v=>(v.s.symptom||v.s.label||'').trim().toLowerCase());if(names[0]&&names[0]===names[1]&&names[1]===names[2]&&sameOna(z.map(v=>v.reiya)))add('haguf','Posible Veset Haguf','El mismo síntoma (“'+(z[2].s.symptom||z[2].s.label)+'”) fue registrado 24 h antes de 3 Reiyot, manteniendo la oná correspondiente.',null,z[2].reiya.ona);}
 // Mimaayán Patuaj: continuidad automática Reiyá → Hefsek, como ya estaba definido.
 if(re.length>=3){
   let x=re.slice(-3),h=x.map(r=>hebParts(atNoon(r.date)));
   if(h.every(Boolean)&&sameOna(x)&&h[0].day===h[1].day&&h[2].day<h[1].day){
     let D=h[1].day,third=x[2];
     let linked=state.events.filter(e=>e.type==='hefsek'&&e.reiyaId===third.id).sort((a,b)=>a.date.localeCompare(b.date))[0];
     let fallback=state.events.filter(e=>e.type==='hefsek'&&e.date>=third.date).sort((a,b)=>a.date.localeCompare(b.date))[0];
     let hef=linked||fallback||null,end=hef?atNoon(hef.date):new Date();end.setHours(12,0,0,0),touches=false;
     for(let q=atNoon(third.date);q<=end;q.setDate(q.getDate()+1)){let z=hebParts(q);if(z&&z.day===D){touches=true;break}}
     if(touches)add('mimaayan','Posible Veset Mimaayán Patuaj','Dos ciclos anteriores comenzaron el día hebreo '+D+'; en el último, la Reiyá comenzó el día '+h[2].day+' y la continuidad hasta el Hefsek volvió a tocar el día '+D+'.',null,third.ona);
   }
 }
 return out;
}
