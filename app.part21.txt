
(function kavuaPersistenceAndJoserBeneor(){
 const basePossible=possiblePatterns,baseAuto=autoEvents,baseRenderPatterns=renderPatterns;
 function reiyot(){return state.events.filter(e=>e.type==='reiya').sort((a,b)=>a.date.localeCompare(b.date)||a.ona.localeCompare(b.ona))}
 function establishmentHistory(){const re=reiyot(),found=[];for(let i=2;i<re.length;i++){const x=re.slice(i-2,i+1),h=x.map(r=>hebParts(atNoon(r.date)));if(h.every(Boolean)&&sameOna(x)&&consecutiveHebMonths(x)&&h[0].day===h[1].day&&h[1].day===h[2].day)found.push({type:'jodesh-kavua',kind:'jodesh',ona:x[2].ona,targetDay:h[2].day,establishedAt:x[2].date,source:x});const wd=x.map(r=>atNoon(r.date).getDay()),ints=[1+dayDiff(x[0].date,x[1].date),1+dayDiff(x[1].date,x[2].date)];if(sameOna(x)&&wd[0]===wd[1]&&wd[1]===wd[2]&&ints[0]===ints[1]&&[8,15,22,29].includes(ints[0]))found.push({type:'hashavua',kind:'hashavua',ona:x[2].ona,weekday:wd[2],interval:ints[0],establishedAt:x[2].date,source:x})}for(let i=3;i<re.length;i++){const x=re.slice(i-3,i+1),ints=[1+dayDiff(x[0].date,x[1].date),1+dayDiff(x[1].date,x[2].date),1+dayDiff(x[2].date,x[3].date)];if(sameOna(x)&&ints[0]===ints[1]&&ints[1]===ints[2])found.push({type:'haflaga-kavua',kind:'haflaga',ona:x[3].ona,interval:ints[2],establishedAt:x[3].date,source:x})}return found.sort((a,b)=>a.establishedAt.localeCompare(b.establishedAt))}
 function signature(p){return p.kind+'|'+p.ona+'|'+(p.targetDay||'')+'|'+(p.weekday??'')+'|'+(p.interval||'')}
 function matchesReiya(p,r){if(!r||r.ona!==p.ona||r.date<=p.establishedAt)return false;if(p.kind==='jodesh'){const h=hebParts(atNoon(r.date));return !!h&&h.day===p.targetDay}if(p.kind==='hashavua')return atNoon(r.date).getDay()===p.weekday;return false}

 function noEventsFor(p){return state.events.filter(e=>e.type==='kavua_no'&&e.patternType===p.type&&e.ona===p.ona&&e.date>p.establishedAt).sort((a,b)=>a.date.localeCompare(b.date))}
 function lifecycleFor(p,all){const replacement=all.find(q=>q.establishedAt>p.establishedAt&&signature(q)!==signature(p));if(replacement)return {...p,status:'replaced',replacement};const nos=noEventsFor(p);let anchor=p.establishedAt,misses=0,lastNo=null;for(const n of nos){const expected=nextProjection(p,anchor);if(expected&&n.date===expected){misses++;anchor=n.date;lastNo=n;if(misses>=3)break}}if(misses<3)return {...p,status:'active',misses,projection:nextProjection(p,anchor),lastNo};const dormantFrom=lastNo.date,later=reiyot().filter(r=>r.date>dormantFrom);let wake=null;if(p.kind==='haflaga')wake=later.find(r=>r.ona===p.ona&&r.date===nextProjection(p,dormantFrom))||null;else wake=later.find(r=>matchesReiya(p,r))||null;if(wake)return {...p,status:'active',misses:0,joserBeneor:true,wakeDate:wake.date,projection:nextProjection(p,wake.date)};return {...p,status:'dormant',misses:3,dormantFrom}}
 function lifecycle(){const hist=establishmentHistory();if(!hist.length)return {hist,latest:null,states:[]};const states=hist.map(p=>lifecycleFor(p,hist)),latest=states.slice().reverse().find(s=>s.status!=='replaced')||null;return {hist,states,latest}}
 function toPattern(s){if(!s||s.status!=='active'||!s.projection)return null;let title='Posible Veset Kavúa',detail='Kavúa vigente.';if(s.kind==='jodesh'){title='Posible Veset HaJódesh Kavúa';detail='Veset HaJódesh fijo del día '+s.targetDay+', en la misma oná.'}if(s.kind==='hashavua'){title='Posible Veset HaShavúa Kavúa';detail='Veset HaShavúa vigente con intervalo inclusivo fijo de '+s.interval+' días, en la misma oná.'}if(s.kind==='haflaga'){title='Posible Veset Haflagá Kavúa';detail='Veset Haflagá vigente con intervalo fijo de '+s.interval+' días, en la misma oná.'}if(s.misses===1||s.misses===2)detail+=' Lleva '+s.misses+' ausencia'+(s.misses===1?'':'s')+' confirmada'+(s.misses===1?'':'s')+'; sigue vigente hasta 3 ausencias.';if(s.joserBeneor)detail+=' Reactivado inmediatamente por Joser Beneor con una sola Reiyá coincidente.';return {...s,type:s.type,title,detail,ona:s.ona,projection:s.projection,kavuaLifecycle:true}}

 possiblePatterns=function(){const base=basePossible(),life=lifecycle(),s=life.latest,lp=toPattern(s),fixed=['jodesh-kavua','hashavua','haflaga-kavua'];let out=base.filter(p=>!fixed.includes(p.type));if(lp)out.push(lp);else if(s&&s.status==='dormant')out.push({type:s.type,title:'Veset Kavúa dormido · Joser Beneor',detail:'Este Kavúa fue arrancado después de 3 ausencias sin ser reemplazado. Queda dormido y puede reactivarse inmediatamente con una sola Reiyá que vuelva a coincidir con su patrón y oná.',projection:null,ona:s.ona,kavuaDormant:true});return out};
 renderPatterns=function(){baseRenderPatterns()};setTimeout(()=>render(),250);authListeners.push(()=>setTimeout(()=>render(),450));
})();
