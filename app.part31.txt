
// Zmanim visuales por localidad: Netz Hajama y Shekiat Hajama.
// La localidad se elige por usuario desde Configuración y se guarda dentro de state.settings.
// No modifica reglas halájicas ni eventos. Criterio de presentación basado en CalJ.
(function zmanimByLocation(){
 const previousOpenSheet=openSheet;
 const previousSyncControls=syncControls;
 const CITIES={
  cdmx:{label:'CDMX',lat:19.4326,lon:-99.1332,timeZone:'America/Mexico_City'},
  acapulco:{label:'Acapulco',lat:16.8531,lon:-99.8237,timeZone:'America/Mexico_City'},
  cuernavaca:{label:'Cuernavaca',lat:18.9242,lon:-99.2216,timeZone:'America/Mexico_City'},
  valle:{label:'Valle de Bravo',lat:19.1950,lon:-100.1310,timeZone:'America/Mexico_City'}
 };
 function ensureZmanimCity(){state.settings=state.settings||{};if(!CITIES[state.settings.zmanimCity])state.settings.zmanimCity='cdmx';return state.settings.zmanimCity;}
 function rad(x){return x*Math.PI/180} function deg(x){return x*180/Math.PI}
 function norm(x,m){return ((x%m)+m)%m}
 function solarUTC(date,lat,lon,rise){
  const N=Math.floor((Date.UTC(date.getFullYear(),date.getMonth(),date.getDate())-Date.UTC(date.getFullYear(),0,0))/86400000);
  const lngHour=lon/15,t=N+(((rise?6:18)-lngHour)/24),M=(0.9856*t)-3.289;
  let L=M+1.916*Math.sin(rad(M))+0.020*Math.sin(rad(2*M))+282.634;L=norm(L,360);
  let RA=deg(Math.atan(0.91764*Math.tan(rad(L))));RA=norm(RA,360);RA+=(Math.floor(L/90)*90-Math.floor(RA/90)*90);RA/=15;
  const sinDec=0.39782*Math.sin(rad(L)),cosDec=Math.cos(Math.asin(sinDec));
  const cosH=(Math.cos(rad(90.833))-(sinDec*Math.sin(rad(lat))))/(cosDec*Math.cos(rad(lat)));if(cosH>1||cosH< -1)return null;
  let H=rise?360-deg(Math.acos(cosH)):deg(Math.acos(cosH));H/=15;
  const T=H+RA-(0.06571*t)-6.622;return norm(T-lngHour,24);
 }
 function localTime(date,utcHours,kind,city){
  if(utcHours==null)return 'No disponible';
  const raw=Date.UTC(date.getFullYear(),date.getMonth(),date.getDate())+utcHours*3600000;
  const ms=kind==='netz'?Math.floor(raw/60000)*60000:Math.ceil(raw/60000)*60000;
  return new Intl.DateTimeFormat('es-MX',{hour:'numeric',minute:'2-digit',hour12:true,timeZone:city.timeZone}).format(new Date(ms));
 }
 function selectedCity(){return CITIES[ensureZmanimCity()]||CITIES.cdmx;}
 function ensureCityControl(){
  ensureZmanimCity();
  const slot=document.getElementById('settingsPersonal');if(!slot)return;
  let card=document.getElementById('zmanimCityCard');
  if(!card){
   card=document.createElement('div');card.id='zmanimCityCard';card.style.cssText='margin:12px 0;padding:14px;border:1px solid #dce9e7;border-radius:14px;background:#f7fbfa';
   card.innerHTML='<div style="font-weight:850;color:#173634;margin-bottom:4px">Localidad para Netz y Shekiat Hajama</div><div style="font-size:13px;color:#66817e;margin-bottom:9px">Los horarios de Onat Yom y Onat Laila se calcularán según esta localidad.</div><select id="zmanimCitySelect" style="width:100%;padding:10px 12px;border:1px solid #cfe0dd;border-radius:10px;background:white;color:#173634;font:inherit"></select>';
   slot.prepend(card);
   const select=card.querySelector('#zmanimCitySelect');
   Object.entries(CITIES).forEach(([key,city])=>{const op=document.createElement('option');op.value=key;op.textContent=city.label;select.appendChild(op)});
   select.onchange=()=>{state.settings.zmanimCity=select.value;save();};
  }
  const select=document.getElementById('zmanimCitySelect');if(select)select.value=state.settings.zmanimCity;
 }
 function hebrewCivilDate(date){return new Date(date.getFullYear(),date.getMonth(),date.getDate(),12,0,0);}
 function removeLegacySunset(){
  const sub=document.getElementById('selSub');if(sub){[...sub.querySelectorAll('div')].forEach(el=>{if(/Puesta del sol/i.test(el.textContent||''))el.remove()});}
  const c=document.getElementById('choices');if(c){[...c.children].forEach(el=>{if(/Puesta del sol/i.test(el.textContent||''))el.remove()});}
  document.querySelectorAll('#modal div').forEach(el=>{if(el.children.length===0&&/Puesta del sol/i.test(el.textContent||''))el.remove()});
 }
 function inject(date,ona){
  removeLegacySunset();const c=document.getElementById('choices');if(!c)return;if(c.querySelector('.zmanimBox'))return;
  const city=selectedCity(),box=document.createElement('div');box.className='zmanimBox';box.style.cssText='grid-column:1 / -1;margin:10px 0 14px;padding:11px 13px;border-radius:12px;background:rgba(31,143,133,.07);line-height:1.7';box.innerHTML='<div><b>Netz Hajama:</b> calculando…</div>'+(ona==='yom'?'<div><b>Shekiat Hajama:</b> calculando…</div>':'');c.insertBefore(box,c.firstChild);
  const zdate=hebrewCivilDate(date),rise=solarUTC(zdate,city.lat,city.lon,true),set=solarUTC(zdate,city.lat,city.lon,false);
  box.innerHTML='<div><b>Netz Hajama:</b> '+localTime(zdate,rise,'netz',city)+'</div>'+(ona==='yom'?'<div><b>Shekiat Hajama:</b> '+localTime(zdate,set,'shekia',city)+'</div>':'')+'<div style="font-size:12px;color:#66817e;margin-top:2px">'+city.label+'</div>';
 }
 openSheet=function(d,o){const result=previousOpenSheet(d,o);setTimeout(()=>{removeLegacySunset();inject(new Date(d.getFullYear(),d.getMonth(),d.getDate(),12,0,0),o)},0);return result;};
 syncControls=function(){previousSyncControls();ensureCityControl();};
 setTimeout(ensureCityControl,350);authListeners.push(()=>setTimeout(ensureCityControl,450));
})();
