function alertHalacha(a){
 if(a.type==='pattern'){let p=a.pattern||{};return '<b>Patrón Kavúa detectado:</b> '+escapeHtml(p.title||'Kavúa')+'. Esta es la oná proyectada por el patrón. Confirma abajo si hubo Reiyá. <span class="notice">Seguimiento matemático; consultar Rab/Posek.</span>'; }
 if(a.label.startsWith('Oná Beinonit'))return '<b>Halajá:</b> durante esta Oná la pareja debe separarse. Deberá revisarse con un <i>Ed</i> (Bediká). Si pasó la Oná y no se revisó, se encuentra <b>asurá (prohibida)</b> hasta que se revise y se encuentre limpia.';
 if(a.label==='Onat HaJódesh')return '<b>Halajá:</b> corresponde a Yom HaJódesh, en la misma fecha hebrea y la misma Oná de la Reiyá anterior. Durante esta Oná debe cuidarse la separación y realizar la Bediká correspondiente.';
 if(a.label.startsWith('Haflagá'))return '<b>Halajá:</b> corresponde al intervalo de Haflagá calculado desde las dos últimas Reiyot. Durante esta Oná debe cuidarse la separación y la Bediká correspondiente.';
 if(a.label.startsWith('Or Zarúa'))return '<b>Aviso:</b> Or Zarúa corresponde a la <b>oná inmediatamente anterior (aprox. 12 horas antes)</b> del jashash indicado. Se configura por separado para Yom HaJódesh, Oná Beinonit y Haflagá.';
 if(a.label.startsWith('Creti Upleti'))return '<b>Aviso:</b> corresponde al cuidado adicional del día 31, 24 horas después del día 30 y manteniendo la oná correspondiente.';
 return 'Aviso de separación según el cálculo registrado.';
}
function getCycleContext(date){
 // Devuelve la última Reiyá previa/igual y el Hefsek asociado (si existe).
 const reiyot=state.events.filter(e=>e.type==='reiya'&&e.date<=date).sort((a,b)=>a.date.localeCompare(b.date));
 const r=reiyot.length?reiyot[reiyot.length-1]:null;
 if(!r)return {reiya:null,hefsek:null,inBleedingWindow:false};
 const nextReiya=state.events.filter(e=>e.type==='reiya'&&e.date>r.date).sort((a,b)=>a.date.localeCompare(b.date))[0]||null;
 let h=state.events.filter(e=>e.type==='hefsek' && ((r.id&&e.reiyaId===r.id)||(!e.reiyaId&&e.date>=r.date&&(!nextReiya||e.date<nextReiya.date))))
   .sort((a,b)=>a.date.localeCompare(b.date))[0]||null;
 // Desde la Reiyá hasta el Hefsek inclusive, el sistema considera continuidad del sangrado.
 // Si todavía no existe Hefsek, continúa bloqueado desde la Reiyá hacia adelante.
 const inWindow=date>=r.date && (!h || date<=h.date) && (!nextReiya || date<nextReiya.date);
 return {reiya:r,hefsek:h,inBleedingWindow:inWindow};
}
function isHefsekOna(date,ona){
 return ona==='yom' && state.events.some(e=>e.type==='hefsek'&&e.date===date&&e.ona==='yom');
}
