 const CITIES={

 }
 function dayOfYear(d){const start=new Date(d.getFullYear(),0,0,12);return Math.floor((new Date(d.getFullYear(),d.getMonth(),d.getDate(),12)-start)/86400000)}
 function sunsetMinutes(d,city){
  const n=dayOfYear(d),gamma=2*Math.PI/365*(n-1);
  const eq=229.18*(0.000075+0.001868*Math.cos(gamma)-0.032077*Math.sin(gamma)-0.014615*Math.cos(2*gamma)-0.040849*Math.sin(2*gamma));
  const decl=0.006918-0.399912*Math.cos(gamma)+0.070257*Math.sin(gamma)-0.006758*Math.cos(2*gamma)+0.000907*Math.sin(2*gamma)-0.002697*Math.cos(3*gamma)+0.00148*Math.sin(3*gamma);
  const lat=city.lat*Math.PI/180;
  const cosH=(Math.cos(90.833*Math.PI/180)/(Math.cos(lat)*Math.cos(decl)))-Math.tan(lat)*Math.tan(decl);
  if(cosH<-1||cosH>1)return null;
  const ha=Math.acos(cosH)*180/Math.PI;
  return 720-4*city.lon-eq+city.tz*60+4*ha;
 }
 function formatClock(minutes){
  if(minutes==null)return 'No disponible';
  minutes=((minutes%1440)+1440)%1440;
  let h=Math.floor(minutes/60),m=Math.round(minutes%60);
  if(m===60){h=(h+1)%24;m=0}
  const suffix=h>=12?'p. m.':'a. m.',h12=h%12||12;
  return h12+':'+String(m).padStart(2,'0')+' '+suffix;
 }
 function civilTitle(d){
  let s=d.toLocaleDateString('es-MX',{weekday:'long',day:'numeric',month:'long'});
  return s.charAt(0).toUpperCase()+s.slice(1);
 }
 function hebTitle(d,ona){
  const hp=hebParts(d);if(!hp)return ona==='yom'?'Onat Yom':'Onat Laila';
  return (ona==='yom'?'Onat Yom':'Onat Laila')+' · '+hp.day+' de '+hMonthEs(hp.month)+' de '+hp.year;
 }

  const lines=[];

  }
  if(lines.length){info.innerHTML=lines.join('');choices.parentNode.insertBefore(info,choices)}
 }
